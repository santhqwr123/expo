
import React, { useState } from "react"; import { Button } from "@/components/ui/button"; import { Card, CardContent } from "@/components/ui/card";

const preguntas = [ "¿Cuál fue tu peor apodo en la escuela?", "¿Alguna vez te enamoraste de un profesor o profesora?", "¿Qué moda seguías que hoy te da vergüenza?", "¿Te copiaron o copiaste en una prueba? ¿Cómo lo hiciste?", "¿Cuál fue tu peor corte de pelo?", "¿Alguna vez mentiste para zafar de una salida?", "¿Tuviste alguna pelea tonta con un amigo que hoy te da risa?", "¿Qué era lo más ridículo que hacías para llamar la atención?", "¿Te gustaba alguien y te rechazó de forma épica?", "¿Cuál fue tu peor papelón en una fiesta?" ];

const retos = [ "Contá una anécdota vergonzosa.", "Imitá a un profesor que te caía mal.", "Hablá 30 segundos con acento chileno.", "Decí una confesión que nunca contaste.", "Hacé una mini coreografía ridícula." ];

export default function JuegoPreguntas() { const [pregunta, setPregunta] = useState(""); const [mostrarReto, setMostrarReto] = useState(false); const [reto, setReto] = useState("");

const sacarPregunta = () => { const index = Math.floor(Math.random() * preguntas.length); setPregunta(preguntas[index]); setMostrarReto(false); };

const noResponder = () => { const index = Math.floor(Math.random() * retos.length); setReto(retos[index]); setMostrarReto(true); };

return ( <div className="flex flex-col items-center justify-center h-screen p-4 bg-pink-50"> <h1 className="text-3xl font-bold mb-6">¿Te Acordás Cuando...?</h1>

<Card className="w-full max-w-md mb-4 text-center">
    <CardContent className="p-6 text-xl">
      {pregunta || "Tocá el botón para sacar una pregunta."}
    </CardContent>
  </Card>

  <div className="flex gap-4">
    <Button onClick={sacarPregunta}>¡Sacar Pregunta!</Button>
    {pregunta && (
      <Button variant="secondary" onClick={noResponder}>No quiero responder</Button>
    )}
  </div>

  {mostrarReto && (
    <div className="mt-6 text-center text-lg font-medium text-red-600">
      Reto: {reto}
    </div>
  )}
</div>

); }
